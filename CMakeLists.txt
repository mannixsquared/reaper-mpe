cmake_minimum_required(VERSION 3.21)
project(reaper_mpe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/reaper-sdk/sdk" CACHE PATH "REAPER SDK headers")
if(NOT EXISTS "${SDK_DIR}/reaper_plugin.h")
  message(FATAL_ERROR "Run python scripts/bootstrap.py before configuring")
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  add_library(reaper_mpe_editor SHARED src/plugin.mm)
  target_compile_options(reaper_mpe_editor PRIVATE -fno-objc-arc)
  target_link_libraries(reaper_mpe_editor PRIVATE "-framework Cocoa")
  add_custom_command(TARGET reaper_mpe_editor POST_BUILD
    COMMAND codesign --force --sign - $<TARGET_FILE:reaper_mpe_editor>)
elseif(WIN32)
  add_library(reaper_mpe_editor SHARED src/plugin_win.cpp)
  target_compile_definitions(reaper_mpe_editor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(reaper_mpe_editor PRIVATE comctl32 gdi32 user32)
else()
  message(FATAL_ERROR "The REAPER editor overlay supports macOS and Windows")
endif()
set_target_properties(reaper_mpe_editor PROPERTIES PREFIX "")
target_include_directories(reaper_mpe_editor PRIVATE "${SDK_DIR}" src)
include(CTest)
if(BUILD_TESTING)
  foreach(name curve vibrato midi_batch)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_include_directories(${name}_test PRIVATE src)
    add_test(NAME ${name} COMMAND ${name}_test)
    if(MSVC)
      target_compile_options(${name}_test PRIVATE /UNDEBUG)
    else()
      target_compile_options(${name}_test PRIVATE -UNDEBUG)
    endif()
  endforeach()
  if(WIN32)
    add_executable(windows_load_test tests/windows_load_test.cpp)
    target_compile_options(windows_load_test PRIVATE /UNDEBUG)
    add_test(NAME windows_load COMMAND windows_load_test $<TARGET_FILE:reaper_mpe_editor>)
  endif()
  if(APPLE)
    foreach(name keyboard_gate plugin_host)
      add_executable(${name}_test tests/${name}_test.mm)
      target_include_directories(${name}_test PRIVATE src "${SDK_DIR}")
      target_compile_options(${name}_test PRIVATE -fno-objc-arc -UNDEBUG)
      target_link_libraries(${name}_test PRIVATE "-framework Cocoa")
    endforeach()
    add_test(NAME keyboard_gate COMMAND keyboard_gate_test)
    add_test(NAME plugin_host COMMAND plugin_host_test $<TARGET_FILE:reaper_mpe_editor>)
  endif()
endif()
